\begin{frame}{Il fattore $J$}
    Il flusso di raggi $\gamma$ da annichilazione dipende dalla fisica delle particelle e dalla distribuzione astrofisica di DM:
    \[
    \frac{d\Phi_\gamma}{d\Omega\,dE}
    =
    \underbrace{\frac{1}{8\pi M_{DM}^2}\sum_f \langle\sigma v\rangle_f \frac{dN_{f\to\gamma}}{dE}}_{\text{fisica delle particelle}}
    \;\times\;
    \underbrace{\int_{\text{l.o.s.}} ds\ \rho_{DM}^2(\vec r)}_{\displaystyle J\text{-factor}}
    \]
    \begin{itemize}
        \item Il fattore $J$ raccoglie tutta l'informazione astrofisica (dipende dal quadrato della densit\`a $\Rightarrow$ regioni dense sono privilegiate)
        \item Target ideali: alta densit\`a di DM, vicini, background astrofisico basso
    \end{itemize}
\end{frame}

\begin{frame}{Il buco nero centrale e gli spikes di DM}
    \begin{itemize}
        \item Al centro della Via Lattea: buco nero supermassiccio Sgr A*, $M = (2.6\pm0.2)\times10^6\,M_\odot$ \paper{Eckart \& Genzel 1997; Ghez et al.\ 1998}
        \item Se la DM \`e presente nella regione centrale ($r \lesssim 0.2$ pc), il buco nero ne ridistribuisce le orbite $\Rightarrow$ \textbf{spike}
        \item L'annichilazione $\propto \rho_{DM}^2$: uno spike amplifica fortemente il segnale atteso
    \end{itemize}
    \vspace{6pt}
    \begin{block}{Crescita adiabatica}
        Il buco nero cresce su tempi $\gg$ tempo orbitale delle particelle di DM $\Rightarrow$ si conservano il momento angolare e l'azione radiale: la DM non viene creata, solo redistribuita.
    \end{block}
\end{frame}

\begin{frame}{Relazione di Gondolo-Silk}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            Halo iniziale con profilo a legge di potenza $\rho \propto r^{-\gamma}$: lo spike risultante ha
            \[
            \rho_{sp}(r) \propto r^{-\gamma_{sp}},
            \qquad
            \gamma_{sp} = \frac{9-2\gamma}{4-\gamma}
            \]
            \begin{itemize}
                \item Per NFW ($\gamma=1$): $\gamma_{sp} = 7/3$
                \item Profilo completo NFW + Gondolo-Silk \paper{Gondolo \& Silk 1999}:
            \end{itemize}
            \[
            \rho(r) =
            \begin{cases}
                \rho_{\rm NFW}(R_{sp})\left(\dfrac{r}{R_{sp}}\right)^{-7/3}, & r < R_{sp}\\[4pt]
                \rho_{\rm NFW}(r), & r \geq R_{sp}
            \end{cases}
            \]
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth]{Spike_slope.png}
        \end{column}
    \end{columns}
\end{frame}

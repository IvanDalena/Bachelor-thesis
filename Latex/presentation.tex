\begin{frame}{Obiettivo della tesi}
    \begin{block}{Domanda}
        Quanto le \textbf{incertezze sul profilo di densit\`a} della dark matter (DM) nella Via Lattea
        influenzano i vincoli di \textbf{rivelazione indiretta} ottenuti dal Centro Galattico?
    \end{block}
    \vspace{4pt}
    \begin{itemize}
        \item Il flusso $\gamma$ da annichilazione dipende dal profilo di DM tramite il \emph{fattore $J$} $\propto \int \rho_{DM}^2\,ds$
        \item Profili diversi $\Rightarrow$ vincoli molto diversi sulla sezione d'urto $\langle\sigma v\rangle$
        \item Al centro c'\`e il buco nero Sgr~A*: la sua crescita pu\`o formare una \textbf{spike} di DM che amplifica enormemente il segnale
    \end{itemize}
    \vspace{4pt}
    \centering
    {\small Confrontiamo profili rappresentativi \& benchmark di spike con un semplice modello di reliquia termica.}
\end{frame}

    \begin{columns}[T]

            \begin{itemize}

                \[
                \langle\sigma v\rangle \simeq 3\times10^{-26}\ \mathrm{cm^3/s}

            \end{itemize}

\begin{frame}{Rivelazione indiretta e fattore $J$}
    \begin{columns}[T]
        \begin{column}{0.62\textwidth}
            \small
            Cerchiamo particelle SM prodotte da annichilazione di DM (raggi $\gamma$).
            Il flusso si fattorizza:
            \[
            \frac{d\Phi_\gamma}{d\Omega\,dE}
            =
            \underbrace{\frac{\langle\sigma v\rangle}{8\pi M_{DM}^2}\frac{dN_\gamma}{dE}}_{\text{fisica delle particelle}}
            \times
            \underbrace{\int_{\text{l.o.s.}} \!\! ds\ \rho_{DM}^2}_{\displaystyle J\text{-factor}}
            \]
            \begin{itemize}
                \item $J$ raccoglie \emph{tutta} l'informazione astrofisica
                \item $\propto \rho_{DM}^2 \Rightarrow$ le regioni dense dominano
                \item \textbf{Target ideale: il Centro Galattico}\\
                flusso $>3$ ordini di grandezza rispetto alle dwarf galaxies
            \end{itemize}
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering
            \includegraphics[width=\textwidth]{DMannihilation.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Cap.\ 2 --- Profili di DM al Centro Galattico}
    Il profilo di DM nella regione interna \`e \textbf{incerto}: le simulazioni non lo risolvono.
    Consideriamo profili rappresentativi, dai pi\`u \emph{piccati} ai pi\`u \emph{cored}:
    \[
    \text{Einasto (piccato)} \quad\longleftrightarrow\quad \text{NFW} \quad\longleftrightarrow\quad \text{Burkert (cored)}
    \]
    Parametri fissati da $\rho(r_\odot)=0.4~\mathrm{GeV/cm^3}$ e $M(<60~\mathrm{kpc})=4.7\times10^{11}M_\odot$.
    \vspace{2pt}
    \begin{table}
    \centering
    \small
    \begin{tabular}{c|c|c|c|c}
    \hline
     & Einasto & Burkert & $\mathrm{NFW}_{30\,\mathrm{pc}}$ & $\mathrm{NFW}_{3\,\mathrm{kpc}}$ \\
    \hline
    $\log_{10}\!\big(J_{\Omega_{\mathrm{obs}}}\,[\mathrm{GeV^2/cm^5}]\big)$
    & 21.96 & 19.35 & 21.57 & 19.77 \\
    \hline
    \end{tabular}
    \end{table}
    \begin{itemize}
        \item $J$ integrato vicino al GC: \textbf{$>2$ ordini di grandezza} tra Einasto e Burkert
        \item $\langle\sigma v\rangle_{\rm lim}\propto 1/J$: il limite Burkert \`e $\sim 4\times10^2$ volte pi\`u debole di Einasto
    \end{itemize}
\end{frame}

\begin{frame}{Cap.\ 3 --- Il buco nero centrale e la spike}
    \begin{itemize}
        \item Al centro della Via Lattea: SMBH \textbf{Sgr~A*}, $M=(4.15\pm0.01)\times10^6\,M_\odot$ \paper{GRAVITY 2019}
        \item La \textbf{crescita adiabatica} del BH (tempi $\gg$ tempo orbitale della DM) ne redistribuisce le orbite $\Rightarrow$ cuspide densa: la \textbf{spike}
        \item Annichilazione $\propto\rho_{DM}^2$: la spike pu\`o amplificare enormemente il segnale
    \end{itemize}
    \vspace{4pt}
    \begin{block}{Idea chiave della crescita adiabatica}
        Si conservano momento angolare e azione radiale $\Rightarrow$ $f'(E',L')=f(E,L)$:
        la DM \textbf{non viene creata}, solo redistribuita verso il centro.
    \end{block}
\end{frame}

\begin{frame}{Cap.\ 3 --- Pendenza della spike: Gondolo--Silk}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \small
            Da un halo $\rho\propto r^{-\gamma}$ la spike ha $\rho_{sp}\propto r^{-\gamma_{sp}}$ con
            \[
            \boxed{\ \gamma_{sp}=\frac{9-2\gamma}{4-\gamma}\ }
            \]
            \begin{itemize}
                \item NFW ($\gamma=1$) $\Rightarrow \gamma_{sp}=7/3$
                \item Halo cored $\Rightarrow$ spike pi\`u \emph{mite} ($\gamma_{sp}\!\to\!3/2$)
                \item La spike \`e sempre molto pi\`u ripida della cuspide iniziale
                \item \textbf{Saturazione} da annichilazione: densit\`a massima
                $\rho_{\rm core}=\dfrac{m_{DM}}{\langle\sigma v\rangle\,t_{BH}}$
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\textwidth]{Spike_slope.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Cap.\ 3 --- Due benchmark di spike}
    \begin{columns}[T]
        \begin{column}{0.44\textwidth}

\begin{frame}{Cap.\ 3 --- Amplificazione del fattore $J$}
    $J$-factor sul pixel centrale $0.1^\circ\times0.1^\circ$ su Sgr~A* (limite non-annichilante):
    \begin{table}
    \centering
    \small
    \begin{tabular}{c|c|c|c||c|c|c}
    \hline
    $\log_{10}J$ & NFW & NFW$+$GS & NFW$+$LSH & Core & Core$+$GS & Core$+$LSH \\
    \hline
    & 20.70 & \textbf{27.74} & 25.22 & 18.63 & 22.87 & 20.53 \\
    \hline
    \end{tabular}
    \end{table}
    \vspace{4pt}
    \begin{itemize}
        \item Enorme aumento per l'halo NFW: fino a $\sim\!7$ ordini di grandezza
        \item GS amplifica pi\`u di LSH; l'halo cored molto meno del piccato
    \end{itemize}
    \begin{block}{Gerarchia delle incertezze}
        Profilo interno dell'halo ($\sim\!5$ ordini tra i due halo) $\gg$ forma della spike ($\sim\!2$ ordini tra i benchmark).
    \end{block}
\end{frame}

\begin{frame}{Risultato principale --- Impatto della spike sui limiti}
    \begin{columns}[T]
        \begin{column}{0.44\textwidth}
            \small
            Limiti H.E.S.S.\ su Sgr~A* con spike \paper{Balaji et al.\ 2023}, vs.\ la nostra reliquia termica.
            \begin{itemize}
                \item \textbf{GS}: limite fino a $\sim\!1.2\times10^{-30}~\mathrm{cm^3/s}$, $>\!4$ ordini sotto il caso senza spike
                \item[$\Rightarrow$] DM termica \textbf{esclusa} per $m_{DM}\gtrsim 3.5$~TeV
                \item \textbf{LSH}: limite pi\`u debole $\Rightarrow$ \textbf{consentita} a tutte le masse
            \end{itemize}
        \end{column}
        \begin{column}{0.56\textwidth}

\end{frame}

\begin{frame}{Conclusioni}
    \begin{itemize}
        \item I vincoli su $\langle\sigma v\rangle$ dipendono \textbf{criticamente} dal profilo interno di DM e dal benchmark di spike assunto
        \item Persino fissando un halo NFW, la sola scelta della spike (GS vs.\ LSH) cambia \emph{qualitativamente} la conclusione
        \item Se esiste DM termica con $m_{DM}\gtrsim 3.5$~TeV, la combinazione NFW$+$GS \`e esclusa: l'halo interno \`e pi\`u piatto, oppure la spike \`e attenuata, oppure non si \`e mai formata
    \end{itemize}
    \vspace{4pt}
    \begin{block}{Messaggio}
        Le osservazioni del Centro Galattico sondano non solo le propriet\`a della DM,
        ma anche la \textbf{storia dinamica} della regione pi\`u interna della Via Lattea.
    \end{block}
